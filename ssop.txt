Proc SearchFile
lpara _seek,_str

	retu _seek
endif
_seek=justfname(_seek)
_str=ntoc(_str)
_str=addpipe(icase(empty(_str),'X',_str))
for ii=1 to getwordcount(_str,'|')
	_folder=ntoc(getwordnum(_str,ii,'|'))
	_folder=calfolder(icase(_folder='X','',_folder))
	_file=ntoc(_folder)+_seek
	_file=sys(2000,_file)
	if !empty(_file)
		_file=_folder+_file
		exit
	endif
next
retu _file
