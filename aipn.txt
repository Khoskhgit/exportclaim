			other

		m.status=icase(empty(m.errorcode),'A','')
		lcan=.T.		&&icase(isnull(m.olderror),.T.,m.status='A',.T.,.F.)
		if !empty(m.id) and lcan
			if m.an='ODS'
				=sqlexec(_handle,[update ods_register set flag='1' where id=?m.an])
			endif
			if m.tcode='C'
				=sqlexec(_handle,textmerge([replace into <<myrcm>>.chi_send (seq,flag) values (?m.an,'1')]))
			endif
			if !savedata2mysql(_table)
				=writeerror(_msg,oerr)
				lwrok=.F.
				exit
			endif
		endif
		n=n+1
		nn=nn+icase(m.tcode='A',1,0)
		if inkey()#0
			lwork=.F.
			exit
		endif
	endif
	=writeas(trans(nn)+'/'+trans(n)+'/'+trans(rec),oErr,2)
next
